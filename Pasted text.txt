function getStoredChats() {

      localStorage.getItem(STORAGE) || "[]"
    );

  } catch {

    return [];

  }

function saveChats(chats) {

  localStorage.setItem(
    STORAGE,
    JSON.stringify(chats)
  );

}


function saveCurrentChat() {

  if (!messages.length) {
    return;
  }

  const chats =
    getStoredChats();

  const title =
    messages.find(
      m => m.role === "user"
    )?.content
    ?.slice(0, 50)
    || "New chat";

  chats.unshift({

    id:
      Date.now(),

    userId,

    title,

    messages

  });

  const filtered =
    chats.filter(
      chat =>
        chat.userId === userId
    );

  saveChats(
    filtered.slice(0, 50)
  );

  renderHistory();

}

              chat.messages

              ? chat.messages
              : [];

          renderMessages();

          sidebar.classList.remove(
            "open"
          );

        }
      );

      history.appendChild(item);

    }
  );

}

/* =========================================
   FILE SIZE
========================================= */

function formatFileSize(bytes) {

  if (bytes < 1024) {

    return bytes + " B";

  }

  if (bytes < 1024 * 1024) {

    return (
      (bytes / 1024)
        .toFixed(1) +
      " KB"
    );

  }

  return (
    (bytes / (1024 * 1024))
      .toFixed(1) +
    " MB"
  );

}

const SpeechRecognition =
  window.SpeechRecognition ||
  window.webkitSpeechRecognition;

let recognition =
  null;

let isListening =
  false;

if (SpeechRecognition) {

  recognition =
    new SpeechRecognition();

  recognition.continuous =
    false;

  recognition.interimResults =
    true;

  recognition.lang =
    "en-US";


  recognition.onstart =
    () => {

      isListening =
        true;

      micButton.classList.add(
        "listening"
      );

      micButton.textContent =
        "🔴";

      status.textContent =
        "Listening...";

    };


  recognition.onresult =
    event => {

      let finalText =
        "";

      let interimText =
        "";

      for (
        let i = event.resultIndex;
        i < event.results.length;
        i++
      ) {

        const transcript =
          event.results[i][0]
            .transcript;

        if (
          event.results[i].isFinal
        ) {

          finalText +=
            transcript;

        } else {

          interimText +=
            transcript;

        }

      }

      if (finalText) {

        input.value =
          (
            input.value
              ? input.value + " "
              : ""
          ) +
          finalText;

      }

      input.style.height =
        "auto";

      input.style.height =
        Math.min(
          input.scrollHeight,
          150
        ) + "px";

    };


  recognition.onerror =
    event => {

      console.error(
        "Speech recognition error:",
        event.error
      );

      status.textContent =
        "Voice input error. Please try again.";

    };


  recognition.onend =
    () => {

      isListening =
        false;

      micButton.classList.remove(
        "listening"
      );

      micButton.textContent =
        "🎤";

      setTimeout(
        () => {

          if (!isListening) {

            status.textContent =
              "";

          }

        },
        1500
      );

    };


  micButton.addEventListener(
    "click",
    () => {

      if (isListening) {

        recognition.stop();

        return;

      }

      try {

        recognition.start();

      } catch (error) {

        console.error(error);

      }

    }
  );

} else {

  micButton.addEventListener(
    "click",
    () => {

      status.textContent =
        "Voice input is not supported by this browser.";

    }
  );

}

async function readFileAsDataURL(file) {
  return new Promise((resolve, reject) => {
    const reader = new FileReader();

    reader.onload = () => resolve(reader.result);
    reader.onerror = () =>
      reject(new Error("Could not read the selected file."));

    reader.readAsDataURL(file);
  });
}

async function sendMessage() {
  const text = input.value.trim();

  if (!text && !selectedFile) {
    return;
  }

  const fileToSend = selectedFile;

  let displayText = text;

  if (fileToSend) {
    if (displayText) {
      displayText += "\n\n📎 " + fileToSend.name;
    } else {
      displayText = "📎 " + fileToSend.name;
    }
  }

  messages.push({
    role: "user",
    content: displayText
  });

  addMessageToScreen(
    "user",
    displayText
  );

  input.value = "";
  input.style.height = "auto";

  scrollToBottom();

  status.textContent = fileToSend
    ? "Preparing your file..."
    : "Zed AI is thinking...";

  send.disabled = true;

  try {
    let filePayload = null;

    if (fileToSend) {
      const maxFileSize = 15 * 1024 * 1024;

      const allowedTypes = [
        "application/pdf",
        "image/jpeg",
        "image/png",
        "image/webp",
        "image/gif"
      ];

      if (fileToSend.size > maxFileSize) {
        throw new Error(
          "The file is too large. Maximum size is 15 MB."
        );
      }

      if (!allowedTypes.includes(fileToSend.type)) {
        throw new Error(
          "Zed AI currently supports JPG, PNG, WEBP, GIF images and PDF files."
        );
      }

      status.textContent =
        "Reading " + fileToSend.name + "...";

      const dataUrl =
        await readFileAsDataURL(fileToSend);

      filePayload = {
        name: fileToSend.name,
        mimeType: fileToSend.type,
        data: dataUrl
      };
    }

    status.textContent =
      fileToSend
        ? "Zed AI is analyzing your file..."
        : "Zed AI is thinking...";

    const response = await fetch(
      "/api/chat",
      {
        method: "POST",

        headers: {
          "Content-Type": "application/json"
        },

        body: JSON.stringify({
          message: text,
          userId,
          memories,

          conversation:
            messages
              .slice(0, -1)
              .map(message => ({
                role: message.role,
                text: message.content
              })),

          file: filePayload
        })
      }
    );

    const data =
      await response.json();

    if (!response.ok) {
      throw new Error(
        data.error ||
        "Request failed."
      );
    }

    const reply =
      data.reply ||
      data.message ||
      "I couldn't generate a response.";

    messages.push({
      role: "assistant",
      content: reply
    });

    addMessageToScreen(
      "assistant",
      reply
    );

    saveCurrentChat();

    if (fileToSend) {
      selectedFile = null;

      photoInput.value = "";
      fileInput.value = "";

      filePreview.classList.remove(
        "show"
      );
    }

  } catch (error) {
    console.error(error);

    const errorMessage =
      error.message ||
      "Sorry, something went wrong. Please try again.";

    messages.push({
      role: "assistant",
      content: errorMessage
    });

    addMessageToScreen(
      "assistant",
      errorMessage
    );

  } finally {
    status.textContent = "";

    send.disabled = false;

    scrollToBottom();
  }
}

async function generateImage() {

  const prompt =
    input.value.trim();

  if (!prompt) {

    status.textContent =
      "Describe the image you want to create first.";

    input.focus();

    return;

  }


  messages.push({

    role: "user",

    content:
      "🎨 Create image: " +
      prompt

  });


  addMessageToScreen(
    "user",
    "🎨 Create image: " +
    prompt
  );


  input.value =
    "";

  input.style.height =
    "auto";


  status.textContent =
    "Creating your image...";


  send.disabled =
    true;


  try {

    const response =
      await fetch(
        "/api/generate-image",
        {
          method: "POST",

          headers: {
            "Content-Type":
              "application/json"
          },

          body: JSON.stringify({

            prompt,

            userId

          })

        }
      );


    const data =
      await response.json();


    if (!response.ok) {

      throw new Error(
        data.error ||
        "Image generation failed."
      );

    }


    const message =
      document.createElement(
        "div"
      );


    message.className =
      "message assistant";


    const bubble =
      document.createElement(
        "div"
      );


    bubble.className =
      "message-bubble";


    if (data.imageUrl) {

      const img =
        document.createElement(
          "img"
        );

      img.src =
        data.imageUrl;

      img.style.maxWidth =
        "100%";

      img.style.borderRadius =
        "12px";

      img.alt =
        "Generated image";

      bubble.appendChild(
        img
      );


    } else if (data.image) {

      const img =
        document.createElement(
          "img"
        );


      img.src =
        data.image.startsWith(
          "data:"
        )
          ? data.image
          : "data:image/png;base64," +
            data.image;


      img.style.maxWidth =
        "100%";

      img.style.borderRadius =
        "12px";

      img.alt =
        "Generated image";


      bubble.appendChild(
        img
      );


    } else {

      bubble.textContent =
        "The image was generated, but no image data was returned.";

    }


    message.appendChild(
      bubble
    );


    chatContent.appendChild(
      message
    );


    saveCurrentChat();

    scrollToBottom();


  } catch (error) {

    console.error(error);


    const errorMessage =
      error.message ||
      "Image generation failed.";


    messages.push({

      role: "assistant",

      content:
        errorMessage

    });


    addMessageToScreen(
      "assistant",
      errorMessage
    );

  } finally {

    send.disabled =
      false;

    status.textContent =
      "";

    scrollToBottom();

  }

}
